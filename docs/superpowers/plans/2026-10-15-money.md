# Money Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A parents-only «Деньги» tab with the month's mandatory payments (monthly and installments), expenses and income by category and by person, quick entry, push reminders 3 days before and on the day, and a «Скоро платить» tile on Home.

**Architecture:** Three tables (`payments`, `payment_marks`, `money_entries`) with parents-only RLS, synced through the existing Dexie outbox/pull engine (child phones receive no rows). Month views are pure functions over those rows. Reminders run server-side: `pg_cron` calls the `notify` Edge Function daily, which finds due payments and pushes to parents once per payment, month and kind.

**Tech Stack:** React 19, TypeScript, Dexie sync engine, Supabase Postgres (RLS, triggers, pg_cron, pg_net), Deno Edge Function, Vitest, PGlite, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-15-money-design.md`

## Global Constraints

- Parents only: RLS `public.is_parent(household_id)` for read and write on all three tables; the tab and the Home tile only for parents.
- Whole rubles; months as `YYYY-MM`; payment day 1–31, clamped to the last day of a short month.
- Categories, in this order: Обязательные (only from paid payments), Продукты, Дети, Транспорт, Кафе и доставка, Дом, Здоровье, Одежда, Другое.
- Reminders at 10:00 Moscow (07:00 UTC): 3 days before and on the day, only unpaid, only to parents with `notify.payments !== false`.
- Missing tables before `db push`: sync skips them (board Task 1 already does this); the tab stays hidden.
- UI copy Russian; comments plain English; lint stays at 18.

## Review Focus

- A payment on the 31st in February — shown on the 28th/29th and reminded then. Test in Task 3.
- An installment past its last month — gone from the list, no reminder. Test in Task 3 and Task 5.
- A parent marking paid offline and the other parent unmarking online — one mark per payment and month, last write wins, no duplicate «Обязательные». Test in Task 3 (sums from marks keyed by payment+month).
- A child's phone — never receives money rows and never shows the tab. Test in Task 1 (DB) and Task 4 (UI).
- The cron call repeated or forged — each reminder is sent once. Test in Task 5.

```sql
-- Money, for parents only: mandatory payments (monthly or installments) with a mark per month,
-- and expenses and income. Children's phones never receive these rows.
create table public.payments (
  id uuid primary key default gen_random_uuid(),
  household_id uuid not null references public.households (id) on delete cascade,
  title text not null check (length(title) between 1 and 80),
  amount int not null check (amount > 0),
  day int not null check (day between 1 and 31),
  kind text not null check (kind in ('monthly', 'installment')),
  total int check (total is null or total > 0),
  first_month text check (first_month is null or first_month ~ '^\d{4}-(0[1-9]|1[0-2])$'),
  note text not null default '' check (length(note) <= 60),
  created_by uuid references public.members (id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz,
  check (kind = 'monthly' or (total is not null and first_month is not null))
);

create table public.payment_marks (
  payment_id uuid not null references public.payments (id) on delete cascade,
  month text not null check (month ~ '^\d{4}-(0[1-9]|1[0-2])$'),
  household_id uuid not null references public.households (id) on delete cascade,
  paid boolean not null,
  amount int not null check (amount >= 0),
  paid_by uuid references public.members (id) on delete set null,
  paid_on date,
  updated_at timestamptz not null default now(),
  primary key (payment_id, month)
);

create table public.money_entries (
  id uuid primary key default gen_random_uuid(),
  household_id uuid not null references public.households (id) on delete cascade,
  kind text not null check (kind in ('expense', 'income')),
  amount int not null check (amount > 0),
  category text check (category is null or length(category) between 1 and 40),
  member_id uuid references public.members (id) on delete set null,
  on_date date not null,
  note text not null default '' check (length(note) <= 80),
  created_by uuid references public.members (id) on delete set null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  deleted_at timestamptz
);

create trigger payments_touch before insert or update on public.payments for each row execute function public.touch_updated_at();
create trigger payment_marks_touch before insert or update on public.payment_marks for each row execute function public.touch_updated_at();
create trigger money_entries_touch before insert or update on public.money_entries for each row execute function public.touch_updated_at();

create function public.guard_money() returns trigger language plpgsql security definer set search_path = public as $$
begin
  if tg_table_name = 'payment_marks' then
    if not exists (select 1 from public.payments where id = new.payment_id and household_id = new.household_id) then raise exception 'wrong_household'; end if;
    if not public.same_family(new.household_id, new.paid_by) then raise exception 'wrong_household'; end if;
  elsif tg_table_name = 'money_entries' then
    if not public.same_family(new.household_id, new.member_id) then raise exception 'wrong_household'; end if;
  end if;
  return new;
end $$;
create trigger payment_marks_guard before insert or update on public.payment_marks for each row execute function public.guard_money();
create trigger money_entries_guard before insert or update on public.money_entries for each row execute function public.guard_money();

alter table public.payments enable row level security;
alter table public.payment_marks enable row level security;
alter table public.money_entries enable row level security;

create policy "parents read payments" on public.payments for select to authenticated using (public.is_parent(household_id));
create policy "parents add payments" on public.payments for insert to authenticated with check (public.is_parent(household_id));
create policy "parents edit payments" on public.payments for update to authenticated using (public.is_parent(household_id)) with check (public.is_parent(household_id));
create policy "parents read marks" on public.payment_marks for select to authenticated using (public.is_parent(household_id));
create policy "parents add marks" on public.payment_marks for insert to authenticated with check (public.is_parent(household_id));
create policy "parents edit marks" on public.payment_marks for update to authenticated using (public.is_parent(household_id)) with check (public.is_parent(household_id));
create policy "parents read entries" on public.money_entries for select to authenticated using (public.is_parent(household_id));
create policy "parents add entries" on public.money_entries for insert to authenticated with check (public.is_parent(household_id));
create policy "parents edit entries" on public.money_entries for update to authenticated using (public.is_parent(household_id)) with check (public.is_parent(household_id));

revoke all on public.payments, public.payment_marks, public.money_entries from anon;
grant select, insert, update on public.payments, public.payment_marks, public.money_entries to authenticated;

alter table public.member_settings alter column notify set default '{"requests":true,"offers":true,"votes":true,"shop":false,"quiet":true,"stories":true,"payments":true}';
update public.member_settings set notify = notify || '{"payments":true}' where not (notify ? 'payments');

do $$
begin
  if exists (select 1 from pg_publication where pubname = 'supabase_realtime') then
    alter publication supabase_realtime add table public.payments, public.payment_marks, public.money_entries;
  end if;
  -- Daily at 07:00 UTC (10:00 Moscow) the notify function looks for payments to remind about.
  if exists (select 1 from pg_available_extensions where name = 'pg_cron') and exists (select 1 from pg_available_extensions where name = 'pg_net') then
    create extension if not exists pg_cron;
    perform cron.schedule('payments-due', '0 7 * * *', $cron$
      select net.http_post(
        url := 'https://pbipnjlsxkbhdyfebdeo.supabase.co/functions/v1/notify',
        body := '{"table": "payments_due"}'::jsonb,
        headers := '{"Content-Type": "application/json"}'::jsonb)
    $cron$);
  end if;
end $$;
```

```ts
import { CATEGORIES, MANDATORY, type MoneyEntry, type Payment, type PaymentMark } from './types';

const pad = (n: number) => String(n).padStart(2, '0');
export const monthKey = (d: Date) => `${d.getFullYear()}-${pad(d.getMonth() + 1)}`;
export function shiftMonth(key: string, n: number) {
  const [y, m] = key.split('-').map(Number);
  const t = y * 12 + (m - 1) + n;
  return `${Math.floor(t / 12)}-${pad((t % 12) + 1)}`;
}
const MONTHS = ['Январь', 'Февраль', 'Март', 'Апрель', 'Май', 'Июнь', 'Июль', 'Август', 'Сентябрь', 'Октябрь', 'Ноябрь', 'Декабрь'];
const GEN = ['января', 'февраля', 'марта', 'апреля', 'мая', 'июня', 'июля', 'августа', 'сентября', 'октября', 'ноября', 'декабря'];
export const monthTitle = (key: string) => MONTHS[Number(key.slice(5, 7)) - 1];
export const dayTitle = (iso: string) => `${Number(iso.slice(8, 10))} ${GEN[Number(iso.slice(5, 7)) - 1]}`;

const lastDay = (key: string) => { const [y, m] = key.split('-').map(Number); return new Date(Date.UTC(y, m, 0)).getUTCDate(); };
export const dueDate = (p: Payment, month: string) => `${month}-${pad(Math.min(p.day, lastDay(month)))}`;

const index = (key: string) => { const [y, m] = key.split('-').map(Number); return y * 12 + m - 1; };
export function installmentNo(p: Payment, month: string): number | null {
  if (p.kind !== 'installment' || !p.first_month || !p.total) return null;
  const n = index(month) - index(p.first_month) + 1;
  return n >= 1 && n <= p.total ? n : null;
}

export function monthPayments(payments: Payment[], marks: PaymentMark[], month: string) {
  return payments
    .filter((p) => !p.deleted_at && (p.kind === 'monthly' || installmentNo(p, month) !== null))
    .map((p) => {
      const mark = marks.find((x) => x.payment_id === p.id && x.month === month);
      return { p, due: dueDate(p, month), no: installmentNo(p, month), mark, paid: !!mark?.paid, amount: mark?.paid ? mark.amount : p.amount };
    })
    .sort((a, b) => a.due.localeCompare(b.due) || a.p.title.localeCompare(b.p.title, 'ru'));
}
export type MonthPayment = ReturnType<typeof monthPayments>[number];

export function nextUnpaid(list: MonthPayment[], today: string) {
  const open = list.filter((x) => !x.paid);
  return open.find((x) => x.due >= today) ?? open[0];
}

const dayNum = (iso: string) => Date.UTC(Number(iso.slice(0, 4)), Number(iso.slice(5, 7)) - 1, Number(iso.slice(8, 10))) / 86_400_000;
export const daysTo = (today: string, due: string) => dayNum(due) - dayNum(today);
const plural = (n: number, one: string, few: string, many: string) => (n % 10 === 1 && n % 100 !== 11 ? one : [2, 3, 4].includes(n % 10) && ![12, 13, 14].includes(n % 100) ? few : many);
export function dueText(days: number) {
  if (days === 0) return 'сегодня';
  if (days === 1) return 'завтра';
  if (days < 0) return `просрочен на ${-days} ${plural(-days, 'день', 'дня', 'дней')}`;
  return `через ${days} ${plural(days, 'день', 'дня', 'дней')}`;
}

const inMonth = (e: MoneyEntry, month: string) => !e.deleted_at && e.on_date.startsWith(month);

export function spendByCategory(entries: MoneyEntry[], list: MonthPayment[], month: string) {
  const sums = new Map<string, { sum: number; count: number }>();
  const add = (cat: string, n: number) => { const s = sums.get(cat) ?? { sum: 0, count: 0 }; sums.set(cat, { sum: s.sum + n, count: s.count + 1 }); };
  for (const x of list) if (x.paid) add(MANDATORY, x.amount);
  for (const e of entries) if (e.kind === 'expense' && inMonth(e, month)) add(e.category && CATEGORIES.includes(e.category) ? e.category : 'Другое', e.amount);
  return [...sums.entries()].map(([cat, v]) => ({ cat, ...v })).filter((c) => c.sum > 0).sort((a, b) => b.sum - a.sum);
}

export function spendByMember(entries: MoneyEntry[], list: MonthPayment[], month: string) {
  const out = new Map<string, number>();
  const add = (id: string | null | undefined, n: number) => { if (id) out.set(id, (out.get(id) ?? 0) + n); };
  for (const e of entries) if (e.kind === 'expense' && inMonth(e, month)) add(e.member_id, e.amount);
  for (const x of list) if (x.paid) add(x.mark?.paid_by, x.amount);
  return out;
}

export const incomeOf = (entries: MoneyEntry[], month: string) => entries.filter((e) => e.kind === 'income' && inMonth(e, month)).reduce((s, e) => s + e.amount, 0);
```

```ts
// useMoney.ts
import { useLiveQuery } from 'dexie-react-hooks';
import type { LocalDb } from '../../data/db';
import { MONEY_TABLES } from './types';

export function useMoney(db: LocalDb, h: string) {
  return useLiveQuery(async () => {
    const [payments, marks, entries] = await Promise.all([
      db.payments.where('household_id').equals(h).filter((p) => !p.deleted_at).toArray(),
      db.payment_marks.where('household_id').equals(h).toArray(),
      db.money_entries.where('household_id').equals(h).filter((e) => !e.deleted_at).toArray(),
    ]);
    return { payments, marks, entries };
  }, [db, h]);
}

export function useMoneyReady(db: LocalDb): boolean {
  return useLiveQuery(async () => {
    const synced = await db.meta.get('synced_at');
    const missing = ((await db.meta.get('missing_tables'))?.value as string[] | undefined) ?? [];
    return !!synced && !MONEY_TABLES.some((t) => missing.includes(t));
  }, [db]) ?? false;
}
```

```ts
// Which payments to remind about today: three days before and on the day, unpaid ones only.
type Payment = { id: string; household_id: string; title: string; amount: number; day: number; kind: string; total: number | null; first_month: string | null; deleted_at: string | null };
type Mark = { payment_id: string; month: string; paid: boolean };

const pad = (n: number) => String(n).padStart(2, '0');
const lastDay = (key: string) => { const [y, m] = key.split('-').map(Number); return new Date(Date.UTC(y, m, 0)).getUTCDate(); };
const index = (key: string) => { const [y, m] = key.split('-').map(Number); return y * 12 + m - 1; };
const shift = (key: string, n: number) => { const t = index(key) + n; return `${Math.floor(t / 12)}-${pad((t % 12) + 1)}`; };
const inRange = (p: Payment, month: string) => {
  if (p.kind !== 'installment') return true;
  if (!p.first_month || !p.total) return false;
  const n = index(month) - index(p.first_month) + 1;
  return n >= 1 && n <= p.total;
};
const due = (p: Payment, month: string) => `${month}-${pad(Math.min(p.day, lastDay(month)))}`;
const plusDays = (iso: string, n: number) => new Date(Date.UTC(Number(iso.slice(0, 4)), Number(iso.slice(5, 7)) - 1, Number(iso.slice(8, 10)) + n)).toISOString().slice(0, 10);

export function dueReminders(payments: Payment[], marks: Mark[], today: string) {
  const soonDay = plusDays(today, 3);
  const out: Array<{ payment: Payment; month: string; due: string; kind: 'soon' | 'today' }> = [];
  for (const p of payments) {
    if (p.deleted_at) continue;
    for (const [day, kind] of [[today, 'today'], [soonDay, 'soon']] as const) {
      const month = day.slice(0, 7);
      if (!inRange(p, month) || due(p, month) !== day) continue;
      if (marks.some((m) => m.payment_id === p.id && m.month === month && m.paid)) continue;
      out.push({ payment: p, month, due: day, kind });
    }
  }
  return out;
}
export { shift as shiftMonth };
```

```ts
async function paymentsDue(): Promise<Response> {
  const db = createClient(Deno.env.get('SUPABASE_URL')!, serverKey(), { auth: { persistSession: false } });
  const today = new Date(Date.now() + 3 * 3600 * 1000).toISOString().slice(0, 10);
  const { data: payments } = await db.from('payments').select('*').is('deleted_at', null);
  const { data: marks } = await db.from('payment_marks').select('payment_id, month, paid');
  const vapid = { subject: Deno.env.get('VAPID_SUBJECT')!, publicKey: Deno.env.get('VAPID_PUBLIC_KEY')!, privateJwk: JSON.parse(Deno.env.get('VAPID_PRIVATE_JWK')!) };
  let sent = 0;
  for (const r of dueReminders(payments ?? [], marks ?? [], today)) {
    const { error: claimed } = await db.from('push_log').insert({ kind: 'payments_due', ref_id: r.payment.id, version: `${r.month}:${r.kind}` });
    if (claimed) continue;
    const { data: parents } = await db.from('members').select('id').eq('household_id', r.payment.household_id).eq('role', 'parent').is('deleted_at', null).not('user_id', 'is', null);
    const ids = (parents ?? []).map((m) => m.id);
    const { data: settings } = await db.from('member_settings').select('member_id, notify').in('member_id', ids);
    const wanted = ids.filter((id) => (settings ?? []).find((s) => s.member_id === id)?.notify?.payments !== false);
    const msg = compose({ kind: 'payments_due', title: r.payment.title, amount: r.payment.amount, when: r.kind }, today);
    const { data: subs } = await db.from('push_subscriptions').select('endpoint, member_id, p256dh, auth').in('member_id', wanted);
    for (const s of subs ?? []) {
      try { const st = await sendPush(s, { ...msg, quiet: false }, vapid); if (st < 300) sent += 1; } catch { /* next */ }
    }
  }
  return Response.json({ sent });
}
```
